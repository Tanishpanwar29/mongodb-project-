db.products.drop();

for (let i = 1; i <= 50; i++) {
  db.products.insertOne({
    productId: i,
    productName: "Product " + i,
    category: i % 3 === 0 ? "Electronics" : i % 3 === 1 ? "Clothing" : "Grocery",
    brand: i % 2 === 0 ? "Brand A" : "Brand B",
    price: i * 100,
    stock: i * 2,
    rating: (i % 5) + 1,
    inStock: i % 4 !== 0,
    tags: i % 2 === 0 ? ["new", "popular"] : ["sale", "limited"],
    seller: {
      name: "Seller " + i,
      city: i % 2 === 0 ? "Jaipur" : "Delhi"
    },
    createdAt: new Date()
  });
}
